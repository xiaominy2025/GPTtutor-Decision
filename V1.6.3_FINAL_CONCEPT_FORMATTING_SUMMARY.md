```python
def format_concepts_section(content):
    """V1.6.3: Keep concepts in 'Concept: Definition' format, one per line."""
    lines = content.strip().splitlines()
    valid_lines = [line for line in lines if ':' in line and len(line.split(':')[0].strip()) > 2]
    return '\n'.join(valid_lines)
```

## ✅ **PART 3 — Concept Format Validator Added to `test_suite.py`**

```python
def validate_concepts_format(concepts_section: str) -> list:
    warnings = []
    lines = [line.strip() for line in concepts_section.strip().split('\n') if line.strip()]
    
    if len(lines) < 2:
        warnings.append("⚠️ Less than 2 concepts provided.")
    
    for i, line in enumerate(lines):
        if ':' not in line:
            warnings.append(f"❌ Line {i+1} missing colon: '{line}'")
        if line.startswith('-') or line.startswith('*'):
            warnings.append(f"❌ Line {i+1} uses a bullet/dash: '{line}'")
        if len(line.split(':')) > 2:
            warnings.append(f"⚠️ Line {i+1} has multiple colons. Format may be off: '{line}'")
    
    return warnings
```
